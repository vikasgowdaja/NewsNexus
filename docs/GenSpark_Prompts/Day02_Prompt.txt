   ```python
   def normalize_topic(raw_topic: str) -> str:
       cleaned = raw_topic.strip()
       lower_cleaned = cleaned.lower()
       typo_map = {
           "nauruto": "Naruto",
       }
       return typo_map.get(lower_cleaned, cleaned)
   ```
